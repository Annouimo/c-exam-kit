## 函数组件

### 整数、数位与二进制

#### 1. 进制转换与拆分数位

基数2/8/10/16；转出非负；结果不溢出。

函数：int digit_value(char c)
digit_value：合法数字字符0–9/a–f/A–F；如 digit_value('F') 返回15。

函数：long long base_to_dec(char s[], int base)
base_to_dec：base为2/8/10/16；允许前导负号，无0x前缀；返回long long。

函数：void dec_to_base(long long n, int base, char out[])
dec_to_base：n>=0；结果写入out，不返回字符串；out足够大，如char out[65]。

函数：int int_to_digits(long long n, int base, int a[])
int_to_digits：n>=0；高位在前写入a，返回位数；0拆成一位0。

函数：long long digits_to_int(int a[], int len, int base)
digits_to_int：a为高位在前的len位数位，各位小于base；返回整数。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>


int digit_value(char c) {
    if (c >= '0' && c <= '9') return c - '0';
    if (c >= 'a' && c <= 'f') return c - 'a' + 10;
    return c - 'A' + 10;
}

long long base_to_dec(char s[], int base) {
    int i = 0, sign = 1;
    long long value = 0;
    if (s[i] == '-') { sign = -1; i++; }
    for (; s[i]; i++) value = value * base + digit_value(s[i]);
    return sign * value;
}

void dec_to_base(long long n, int base, char out[]) {
    char digits[] = "0123456789ABCDEF";
    char temp[65];
    int len = 0;
    do {
        temp[len++] = digits[n % base];
        n /= base;
    } while (n > 0);
    for (int i = 0; i < len; i++) out[i] = temp[len - 1 - i];
    out[len] = '\0';
}

/* 非负整数拆成高位在前的数位数组；n=0 得到一位 0 */
int int_to_digits(long long n, int base, int a[]) {
    int len = 0;
    do {
        a[len++] = (int)(n % base);
        n /= base;
    } while (n > 0);
    for (int i = 0; i < len / 2; i++) {
        int t = a[i]; a[i] = a[len - 1 - i]; a[len - 1 - i] = t;
    }
    return len;
}

long long digits_to_int(int a[], int len, int base) {
    long long value = 0;
    for (int i = 0; i < len; i++) value = value * base + a[i];
    return value;
}
```

#### 2. 输出指定位数二进制串与循环移动二进制位

unsigned long long按64位；1<=width<=64，out至少width+1；k>=0。

函数：void fixed_binary(unsigned long long x, int width, char out[])；unsigned long long rotate_left64(unsigned long long x,int k)；unsigned long long rotate_right64(unsigned long long x,int k)
fixed_binary(x,32,out)显示低32位；负数可先转unsigned long long。rotate_left64/right64按64位循环，k=0不移64位。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>


void fixed_binary(unsigned long long x, int width, char out[]) {
    for(int i=0;i<width;i++) out[i]=(char)('0'+((x>>(width-1-i))&1ULL));
    out[width]='\0';
}
unsigned long long rotate_left64(unsigned long long x,int k) {
    k%=64;
    return k ? (x<<k)|(x>>(64-k)) : x;
}
unsigned long long rotate_right64(unsigned long long x,int k) {
    k%=64;
    return k ? (x>>k)|(x<<(64-k)) : x;
}
```

#### 3. 读取、修改二进制位及统计1的个数（64位）

用unsigned long long存64位非负数；位号0..63，1ULL用于构造掩码。

函数：int get_bit(unsigned long long x, int pos)
get_bit：unsigned long long输入；pos从0开始，返回0或1。

函数：unsigned long long set_bit(unsigned long long x, int pos)
set_bit：返回修改后的unsigned long long值；使用x=set_bit(x,pos)。

函数：unsigned long long clear_bit(unsigned long long x, int pos)
clear_bit：返回修改后的unsigned long long值；使用x=clear_bit(x,pos)。

函数：unsigned long long toggle_bit(unsigned long long x, int pos)
toggle_bit：返回修改后的unsigned long long值；使用x=toggle_bit(x,pos)。

函数：unsigned long long pow2u(int n)
pow2u：返回1ULL<<n；n小于64。

函数：int is_power_of_two(unsigned long long x)
is_power_of_two：是返回1，否则0；0返回0。

函数：int count_ones(unsigned long long x)
count_ones：如count_ones(13U)返回3。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>


int get_bit(unsigned long long x, int pos) { return (x >> pos) & 1ULL; }

unsigned long long set_bit(unsigned long long x, int pos) { return x | (1ULL << pos); }

unsigned long long clear_bit(unsigned long long x, int pos) { return x & ~(1ULL << pos); }

unsigned long long toggle_bit(unsigned long long x, int pos) { return x ^ (1ULL << pos); }

unsigned long long pow2u(int n) { return 1ULL << n; }

int is_power_of_two(unsigned long long x) {
    return x != 0 && (x & (x - 1)) == 0;
}

int count_ones(unsigned long long x) {
    int count = 0;
    while (x) {
        x &= x - 1;                /* 每次消掉最低位的一个 1 */
        count++;
    }
    return count;
}
```

#### 4. 最大公约数与最小公倍数

不输入LLONG_MIN；LCM不溢出。

函数：long long gcd(long long a, long long b)
gcd：结果非负；gcd(0,0)=0；不输入LLONG_MIN。

函数：long long lcm(long long a, long long b)
lcm：有一个输入为0则返回0；输入不为LLONG_MIN，结果不溢出。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>


long long gcd(long long a, long long b) {
    if (a < 0) a = -a;
    if (b < 0) b = -b;
    while (b) {
        long long t = a % b;
        a = b; b = t;
    }
    return a;
}

long long lcm(long long a, long long b) {
    if (a == 0 || b == 0) return 0;
    if (a < 0) a = -a;
    if (b < 0) b = -b;
    return a / gcd(a, b) * b;
}
```

#### 6. 预先筛出范围内的素数（素数筛）

查询值为0..SIEVE_MAX；建表一次。

函数：void build_primes_sieve(int n)
build_primes_sieve：0<=n<=SIEVE_MAX；调用后查is_prime[x]，素数表为primes[0..pc-1]；x不超过n。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define SIEVE_MAX 1000000

int is_prime[SIEVE_MAX + 1];

int primes[SIEVE_MAX + 1], pc;

void build_primes_sieve(int n) {
    memset(is_prime, 0, sizeof is_prime);
    pc = 0;
    for (int i = 2; i <= n; i++) is_prime[i] = 1;
    for (int i = 2; i <= n / i; i++) {
        if (is_prime[i])
            for (int j = i * i; j <= n; j += i) is_prime[j] = 0;
    }
    for (int i = 2; i <= n; i++)
        if (is_prime[i]) primes[pc++] = i;
}
```

#### 8. 找最长相等段或逐项加1的连续段

consecutive=0找相等元素连续段，=1找后一项比前一项大1的连续段；0下标。

函数：int longest_run(int a[], int n, int consecutive)
longest_run(a,n,0)或longest_run(a,n,1)。修改ok可适配其他相邻关系。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>


int longest_run(int a[], int n, int consecutive) {
    if (n == 0) return 0;
    int length = 1, best = 1;
    for (int i = 1; i < n; i++) {
        int ok = consecutive ? (a[i] > a[i-1] && (unsigned long long)a[i] - (unsigned long long)a[i-1] == 1) : a[i] == a[i-1];
        length = ok ? length+1 : 1;
        if (length > best) best = length;
    }
    return best;
}
```

#### 10. 求前面的元素减后面的元素的最大差

n>=2；差值为前减后；保留负答案。

函数：long long max_forward_diff(int a[], int n)
max_forward_diff：n>=2；求max(a[i]-a[j])且i<j，保留负答案。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define ARRAY_MAX 100010

long long max_forward_diff(int a[], int n) {
    long long max_before = a[0];
    long long best = (long long)a[0] - a[1];
    for (int j = 1; j < n; j++) {
        long long diff = max_before - a[j];
        if (diff > best) best = diff;
        if (a[j] > max_before) max_before = a[j];
    }
    return best;
}
```

#### 11. 多次查询数组某段的和（前缀和）

1下标；和使用long long；数组不修改。

函数：void build_prefix(int a[], int n, long long prefix[])
build_prefix：a[1..n]；prefix至少n+1个long long；设置prefix[0]=0。

函数：long long range_sum(long long prefix[], int left, int right)
range_sum：返回prefix[right]-prefix[left-1]；1<=left<=right<=n。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define ARRAY_MAX 100010

void build_prefix(int a[], int n, long long prefix[]) {
    prefix[0] = 0;
    for (int i = 1; i <= n; i++) prefix[i] = prefix[i - 1] + a[i];
}

/* 闭区间 [left,right]，要求 1<=left<=right<=n */
long long range_sum(long long prefix[], int left, int right) {
    return prefix[right] - prefix[left - 1];
}
```

#### 12. 反复查询某段的最大值（ST表/RMQ）

数组建表后不再修改；先build_rmq，再按0下标闭区间查询最大值。

函数：void build_rmq(int a[], int n)
build_rmq：a[0..n-1]，1<=n<=RMQ_MAX；建表后数组不修改，每组重新建表。

函数：int range_max(int left, int right)
range_max：建表后调用，0<=left<=right<n，闭区间查询；改最小值需同时修改建表与查询。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define RMQ_MAX 100000
#define RMQ_LOG 17

int st[RMQ_LOG][RMQ_MAX];

int log2_floor[RMQ_MAX + 1];

void build_rmq(int a[], int n) {
    log2_floor[1] = 0;
    for (int i = 2; i <= n; i++) log2_floor[i] = log2_floor[i/2] + 1;
    for (int i = 0; i < n; i++) st[0][i] = a[i];
    for (int k = 1; (1 << k) <= n; k++) {
        for (int i = 0; i + (1 << k) <= n; i++) {
            int x = st[k-1][i], y = st[k-1][i + (1 << (k-1))];
            st[k][i] = x > y ? x : y;
        }
    }
}

int range_max(int left, int right) {
    int k = log2_floor[right - left + 1];
    int x = st[k][left], y = st[k][right - (1 << k) + 1];
    return x > y ? x : y;
}
```

#### 13. 两个矩阵相乘

a为n*m，b为m*p，输出n*p；固定列容量，三数组不重叠，中间结果不溢出。

函数：void matrix_multiply(long long a[][MATRIX_MAX], long long b[][MATRIX_MAX],
                     long long result[][MATRIX_MAX], int n, int m, int p)
matrix_multiply(a,b,c,n,m,p)。大矩阵放全局/static；复杂度O(nmp)。

#### 15. 手写数组排序与查找（冒泡、线性、二分）

0下标；二分要求升序；失败-1，重复返回任意匹配。

函数：void swap_int(int *a, int *b)
swap_int：传地址：swap_int(&a,&b)。

函数：void bubble_sort(int a[], int n)
bubble_sort：把a[0..n-1]原地升序排序。

函数：int linear_search(int a[], int n, int key)
linear_search：返回首次匹配的0下标，失败-1。

函数：int binary_search(int a[], int n, int key)
binary_search：a必须升序；返回任意匹配的0下标，失败-1。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>


void swap_int(int *a, int *b) {
    int t = *a; *a = *b; *b = t;
}

void bubble_sort(int a[], int n) {
    for (int i = 0; i < n - 1; i++) {
        int swapped = 0;
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j + 1]) {
                swap_int(&a[j], &a[j + 1]);
                swapped = 1;
            }
        }
        if (!swapped) break;
    }
}

int linear_search(int a[], int n, int key) {
    for (int i = 0; i < n; i++)
        if (a[i] == key) return i;
    return -1;
}

/* 必须升序 */
int binary_search(int a[], int n, int key) {
    int left = 0, right = n - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (a[mid] == key) return mid;
        if (a[mid] < key) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}
```

#### 20. 去行末换行、判断回文与处理子串

按字节处理；key非空。count_substring的overlap为1允许重叠，0不重叠。

函数：void trim_newline(char s[])；int is_palindrome(char s[])；int erase_first(char s[], char key[])；int count_substring(char s[], char key[], int overlap)
trim_newline(s); is_palindrome(s); erase_first(s,key); count_substring(s,key,1)。去换行不会删掉没有换行的最后一个字符；erase_first一次删除，反复调用会重新匹配新串。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>


void trim_newline(char s[]) {
    size_t n = strlen(s);
    if (n && s[n-1] == '\n') s[--n] = '\0';
    if (n && s[n-1] == '\r') s[n-1] = '\0';
}
int is_palindrome(char s[]) {
    size_t n = strlen(s);
    for (size_t i = 0; i < n/2; i++) if (s[i] != s[n-1-i]) return 0;
    return 1;
}
int erase_first(char s[], char key[]) {
    size_t n = strlen(key);
    if (!n) return 0;
    char *p = strstr(s, key);
    if (!p) return 0;
    memmove(p, p+n, strlen(p+n)+1);
    return 1;
}
int count_substring(char s[], char key[], int overlap) {
    size_t n = strlen(key);
    if (!n) return 0;
    int count = 0;
    char *p = s;
    while ((p = strstr(p, key)) != NULL) { count++; p += overlap ? 1 : n; }
    return count;
}
```

#### 30. 求一元二次方程的根

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>


typedef struct {
    int kind;
    double x1, x2, re, im;
} QuadraticResult;

QuadraticResult solve_quadratic(double a, double b, double c, double eps) {
    QuadraticResult r = {0, 0, 0, 0, 0};
    if (fabs(a) <= eps) { r.kind = -1; return r; }
    double delta = b*b - 4*a*c;
    if (fabs(delta) <= eps) {
        r.kind = 1;
        r.x1 = r.x2 = -b / (2*a);
    } else if (delta > 0) {
        r.kind = 2;
        /* 稳定的等价公式，减少 -b 与 sqrt(delta) 相减时的消去误差 */
        double q = -0.5 * (b + (b >= 0 ? sqrt(delta) : -sqrt(delta)));
        r.x1 = q / a;
        r.x2 = c / q;
        if (r.x1 > r.x2) {
            double t = r.x1; r.x1 = r.x2; r.x2 = t;
        }
    } else {
        r.kind = 0;
        r.re = -b / (2*a);
        r.im = sqrt(-delta) / (2*fabs(a));
    }
    return r;
}
```

stack_init(&s); stack_push(&s,ch); stack_top(&s); stack_pop(&s)。brackets_match(text)返回1/0，内部每次重置栈且不按值复制大对象。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define STACK_CAP 10005

typedef struct { char data[STACK_CAP]; int size; } CharStack;
void stack_init(CharStack *s) { s->size=0; }
int stack_empty(CharStack *s) { return s->size==0; }
void stack_push(CharStack *s, char value) { s->data[s->size++]=value; }
char stack_top(CharStack *s) { return s->data[s->size-1]; }
void stack_pop(CharStack *s) { s->size--; }
int brackets_match(char text[]) {
    static CharStack s;
    stack_init(&s);
    for(int i=0; text[i]; i++) {
        char ch=text[i];
        if(ch=='(' || ch=='[' || ch=='{') stack_push(&s,ch);
        else if(ch==')' || ch==']' || ch=='}') {
            if(stack_empty(&s)) return 0;
            char t=stack_top(&s);
            if(!((t=='('&&ch==')')||(t=='['&&ch==']')||(t=='{'&&ch=='}'))) return 0;
            stack_pop(&s);
        }
    }
    return stack_empty(&s);
}
```

#### 42. 判断网格可达性、统计连通区域（DFS）

0表示不能走，1表示能走；四方向移动，每格只访问一次；可查可达性或区域大小，不求最短路。

函数：int dfs_grid(int r, int c)
dfs_grid：先填rows/cols、0/1网格并清visited；返回本次新访问格数，访问标记不撤销；不是最短路。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define GRID_MAX 100

int grid[GRID_MAX][GRID_MAX], visited[GRID_MAX][GRID_MAX];

int rows, cols;

int dr[4] = {1, -1, 0, 0};

int dc[4] = {0, 0, 1, -1};

/* 返回新访问的格子数；越界、墙、已访问均返回 0 */
int dfs_grid(int r, int c) {
    if (r < 0 || r >= rows || c < 0 || c >= cols) return 0;
    if (grid[r][c] != 1 || visited[r][c]) return 0;
    visited[r][c] = 1;
    int count = 1;
    for (int k = 0; k < 4; k++) count += dfs_grid(r + dr[k], c + dc[k]);
    return count;
}
```

#### 43. 统计网格中只向右、向下走的路径数

只允许向右或向下；0下标，先设置障碍，每次自动清路径计数；结果不超过long long。

函数：long long count_grid_paths(int rows, int cols)
memset(blocked,0,sizeof blocked); 设置障碍后count_grid_paths(rows,cols)。每次清dp；起点或终点被阻挡返回0；改转移适配题意。

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define GRID_PATH_CAP 105

int blocked[GRID_PATH_CAP][GRID_PATH_CAP];
long long path_dp[GRID_PATH_CAP][GRID_PATH_CAP];
long long count_grid_paths(int rows, int cols) {
    memset(path_dp,0,sizeof path_dp);
    for(int r=0;r<rows;r++) for(int c=0;c<cols;c++) {
        if(blocked[r][c]) continue;
        if(r==0 && c==0) path_dp[r][c]=1;
        else {
            if(r) path_dp[r][c]+=path_dp[r-1][c];
            if(c) path_dp[r][c]+=path_dp[r][c-1];
        }
    }
    return path_dp[rows-1][cols-1];
}
```
